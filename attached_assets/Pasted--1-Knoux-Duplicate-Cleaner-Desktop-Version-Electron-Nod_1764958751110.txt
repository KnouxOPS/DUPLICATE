🔥 3. Ultra‑Level Examples لكل وظيفة
📁 3.1 Example: Real FS Scan
import fs from "fs";
import path from "path";

export function scanFolder(folder) {
  const files = [];

  for (const item of fs.readdirSync(folder)) {
    const full = path.join(folder, item);
    const stat = fs.statSync(full);

    if (stat.isDirectory()) {
      files.push(...scanFolder(full));
    } else {
      files.push({
        path: full,
        size: stat.size,
        ext: path.extname(full),
        created: stat.birthtimeMs,
        modified: stat.mtimeMs,
      });
    }
  }

  return files;
}

🔐 3.2 Example: Real Hash Engine
import fs from "fs";
import crypto from "crypto";

export function hashFile(path) {
  const buffer = fs.readFileSync(path);
  return crypto.createHash("sha256").update(buffer).digest("hex");
}

📊 3.3 Example: Extract Image Metadata
import sharp from "sharp";

export async function getImageInfo(path) {
  const data = await sharp(path).metadata();
  return {
    width: data.width,
    height: data.height,
    format: data.format,
  };
}

🗑 3.5 Example: Safe Trash
export function moveToTrash(path) {
  const trash = `${process.env.APPDATA}/Knoux/Trash`;
  if (!fs.existsSync(trash)) fs.mkdirSync(trash, { recursive: true });

  const dest = trash + "/" + path.split("\\").pop();
  fs.renameSync(path, dest);

  return dest;
}

⚙ 3.6 Example: Batch Rule – keep_largest
export function ruleKeepLargest(files) {
  const largest = files.reduce((a, b) => (a.size > b.size ? a : b));
  return {
    keep: largest,
    remove: files.filter(f => f !== largest),
  };
}
